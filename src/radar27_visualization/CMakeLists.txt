cmake_minimum_required(VERSION 3.22)
project(radar27_visualization LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(radar27_interfaces REQUIRED)
find_package(rm_field REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui)
find_package(yaml-cpp REQUIRED)
find_package(Open3D REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Core Widgets)

add_library(${PROJECT_NAME}_core SHARED src/radarmap.cpp src/config.cpp src/rviz_visualizer.cpp)
target_include_directories(${PROJECT_NAME}_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME}_core rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs geometry_msgs visualization_msgs tf2 tf2_ros)
target_link_libraries(${PROJECT_NAME}_core ${OpenCV_LIBS} yaml-cpp::yaml-cpp Open3D::Open3D)
install(TARGETS ${PROJECT_NAME}_core EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
add_library(map_node_component SHARED src/map_node.cpp)
target_include_directories(map_node_component PRIVATE include)
ament_target_dependencies(map_node_component rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs geometry_msgs visualization_msgs tf2 tf2_ros)
target_link_libraries(map_node_component ${PROJECT_NAME}_core)
install(TARGETS map_node_component DESTINATION lib)
add_executable(map_node src/map_node.cpp)
target_include_directories(map_node PRIVATE include)
ament_target_dependencies(map_node rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs geometry_msgs visualization_msgs tf2 tf2_ros)
target_link_libraries(map_node ${PROJECT_NAME}_core)
install(TARGETS map_node DESTINATION lib/${PROJECT_NAME})
rclcpp_components_register_nodes(map_node_component "MapNode")
add_library(rviz_debug_node_component SHARED src/rviz_debug_node.cpp)
target_include_directories(rviz_debug_node_component PRIVATE include)
ament_target_dependencies(rviz_debug_node_component rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs geometry_msgs visualization_msgs tf2 tf2_ros)
target_link_libraries(rviz_debug_node_component ${PROJECT_NAME}_core)
install(TARGETS rviz_debug_node_component DESTINATION lib)
add_executable(rviz_debug_node src/rviz_debug_node.cpp)
target_include_directories(rviz_debug_node PRIVATE include)
ament_target_dependencies(rviz_debug_node rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs geometry_msgs visualization_msgs tf2 tf2_ros)
target_link_libraries(rviz_debug_node ${PROJECT_NAME}_core)
install(TARGETS rviz_debug_node DESTINATION lib/${PROJECT_NAME})
rclcpp_components_register_nodes(rviz_debug_node_component "RvizDebugNode")
add_executable(qt_display_node src/qt_display_node.cpp)
target_include_directories(qt_display_node PRIVATE include)
ament_target_dependencies(qt_display_node rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs geometry_msgs visualization_msgs tf2 tf2_ros)
target_link_libraries(qt_display_node ${PROJECT_NAME}_core)
install(TARGETS qt_display_node DESTINATION lib/${PROJECT_NAME})
target_link_libraries(qt_display_node Qt5::Core Qt5::Widgets)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES RVIZ_DEBUG.md DESTINATION share/${PROJECT_NAME}/docs)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include")
 install(DIRECTORY include/ DESTINATION include)
 ament_export_include_directories(include)
endif()
ament_export_dependencies(OpenCV yaml-cpp Open3D rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs geometry_msgs visualization_msgs tf2 tf2_ros)
ament_package()
