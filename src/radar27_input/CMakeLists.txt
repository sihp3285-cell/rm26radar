cmake_minimum_required(VERSION 3.22)
project(radar27_input LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc videoio)

add_library(video_node_component SHARED src/video_node.cpp)
target_include_directories(video_node_component PRIVATE include)
ament_target_dependencies(video_node_component rclcpp rclcpp_components sensor_msgs cv_bridge std_msgs std_srvs)
target_link_libraries(video_node_component ${OpenCV_LIBS})
install(TARGETS video_node_component DESTINATION lib)
add_executable(video_node src/video_node.cpp)
target_include_directories(video_node PRIVATE include)
ament_target_dependencies(video_node rclcpp rclcpp_components sensor_msgs cv_bridge std_msgs std_srvs)
target_link_libraries(video_node ${OpenCV_LIBS})
install(TARGETS video_node DESTINATION lib/${PROJECT_NAME})
rclcpp_components_register_nodes(video_node_component "VideoNode")
option(BUILD_CAMERA "Build the industrial camera backend" OFF)
if(BUILD_CAMERA)
 add_subdirectory(vendor/rb26SDK)
 foreach(target camera_node_component camera_node)
  if(target STREQUAL "camera_node_component")
   add_library(${target} SHARED src/camera_node.cpp)
   install(TARGETS ${target} DESTINATION lib)
  else()
   add_executable(${target} src/camera_node.cpp)
   install(TARGETS ${target} DESTINATION lib/${PROJECT_NAME})
  endif()
  ament_target_dependencies(${target} rclcpp rclcpp_components sensor_msgs cv_bridge std_msgs std_srvs)
  target_link_libraries(${target} rb26SDK ${OpenCV_LIBS})
 endforeach()
 rclcpp_components_register_nodes(camera_node_component "radar27_input::CameraNode")
 install(TARGETS rb26SDK LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include")
 install(DIRECTORY include/ DESTINATION include)
 ament_export_include_directories(include)
endif()
ament_export_dependencies(rclcpp rclcpp_components sensor_msgs cv_bridge std_msgs std_srvs)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
