cmake_minimum_required(VERSION 3.22)
project(radar27_detection LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(radar27_interfaces REQUIRED)
find_package(rm_field REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc dnn)
find_package(yaml-cpp REQUIRED)
enable_language(CUDA)
set(CMAKE_CUDA_STANDARD 17)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
 set(CMAKE_CUDA_ARCHITECTURES 86)
endif()
find_package(CUDAToolkit REQUIRED)
set(TENSORRT_ROOT "" CACHE PATH "TensorRT installation prefix")
find_path(TENSORRT_INCLUDE NvInfer.h HINTS ${TENSORRT_ROOT}/include REQUIRED)
find_library(TENSORRT_LIBRARY nvinfer HINTS ${TENSORRT_ROOT}/lib REQUIRED)
find_library(TENSORRT_PLUGIN_LIBRARY nvinfer_plugin HINTS ${TENSORRT_ROOT}/lib REQUIRED)

add_library(${PROJECT_NAME}_core SHARED src/model.cpp src/pipeline.cpp src/preprocess.cu src/draw.cpp src/config.cpp)
target_include_directories(${PROJECT_NAME}_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME}_core rm_field)
target_link_libraries(${PROJECT_NAME}_core ${OpenCV_LIBS} yaml-cpp::yaml-cpp ${TENSORRT_LIBRARY} ${TENSORRT_PLUGIN_LIBRARY} CUDA::cudart)
install(TARGETS ${PROJECT_NAME}_core EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
add_library(detect_node_component SHARED src/detect_node.cpp)
target_include_directories(detect_node_component PRIVATE include)
ament_target_dependencies(detect_node_component rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs)
target_link_libraries(detect_node_component ${PROJECT_NAME}_core)
install(TARGETS detect_node_component DESTINATION lib)
add_executable(detect_node src/detect_node.cpp)
target_include_directories(detect_node PRIVATE include)
ament_target_dependencies(detect_node rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs)
target_link_libraries(detect_node ${PROJECT_NAME}_core)
install(TARGETS detect_node DESTINATION lib/${PROJECT_NAME})
rclcpp_components_register_nodes(detect_node_component "DetectNode")
target_include_directories(${PROJECT_NAME}_core SYSTEM PUBLIC ${TENSORRT_INCLUDE})

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include")
 install(DIRECTORY include/ DESTINATION include)
 ament_export_include_directories(include)
endif()
ament_export_dependencies(OpenCV yaml-cpp CUDAToolkit rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
