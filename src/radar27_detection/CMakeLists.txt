cmake_minimum_required(VERSION 3.22)
project(radar27_detection LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(radar27_interfaces REQUIRED)
find_package(rm_field REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc dnn videoio)
find_package(Threads REQUIRED)
find_package(yaml-cpp REQUIRED)

enable_language(CUDA)
set(CMAKE_CUDA_STANDARD 17)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
 set(CMAKE_CUDA_ARCHITECTURES 86)
endif()
find_package(CUDAToolkit REQUIRED)
set(TENSORRT_ROOT "" CACHE PATH "TensorRT installation prefix")
find_path(TENSORRT_INCLUDE NvInfer.h HINTS ${TENSORRT_ROOT}/include REQUIRED)
find_library(TENSORRT_LIBRARY nvinfer HINTS ${TENSORRT_ROOT}/lib REQUIRED)
find_library(TENSORRT_PLUGIN_LIBRARY nvinfer_plugin HINTS ${TENSORRT_ROOT}/lib REQUIRED)

add_library(${PROJECT_NAME}_core SHARED src/model.cpp src/pipeline.cpp src/preprocess.cu src/draw.cpp src/config.cpp)
target_include_directories(${PROJECT_NAME}_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME}_core rm_field)
target_link_libraries(${PROJECT_NAME}_core ${OpenCV_LIBS} yaml-cpp::yaml-cpp ${TENSORRT_LIBRARY} ${TENSORRT_PLUGIN_LIBRARY} CUDA::cudart)
install(TARGETS ${PROJECT_NAME}_core EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
add_library(${PROJECT_NAME}_input SHARED src/input/video_source.cpp src/input/camera_source.cpp src/recorder.cpp)
target_include_directories(${PROJECT_NAME}_input PUBLIC
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
 $<INSTALL_INTERFACE:include>
 ${OpenCV_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME}_input PUBLIC ${OpenCV_LIBS} Threads::Threads)
option(BUILD_CAMERA "Build available industrial camera backends" OFF)
if(BUILD_CAMERA)
 find_path(HIK_INCLUDE_DIR MvCameraControl.h HINTS /opt/MVS/include)
 find_library(HIK_LIBRARY MvCameraControl HINTS /opt/MVS/lib/64 /opt/MVS/lib/aarch64)
 find_path(DAHENG_INCLUDE_DIR GxIAPI.h HINTS /opt/Galaxy_camera/inc /usr/local/include)
 find_library(DAHENG_LIBRARY gxiapi HINTS /opt/Galaxy_camera/lib/x86_64 /opt/Galaxy_camera/lib/armv8)
 if(HIK_INCLUDE_DIR AND HIK_LIBRARY)
  # Hik 走本包内 rb26SDK 的 sdk::HikCamera（rb26SDK/src/hik/hik.cpp）。
  # SDK 目录原样使用，不修改其中任何文件；这里先建目标再链接，
  # include 目录、RB26SDK_HAS_HIK 定义和 MvCameraControl 都由该目标导出。
  add_subdirectory(rb26SDK)
  target_compile_definitions(${PROJECT_NAME}_input PRIVATE RADAR27_HAS_HIK=1)
  target_link_libraries(${PROJECT_NAME}_input PRIVATE rb26SDK)
  message(STATUS "CameraSource: Hik via rb26SDK (hik.cpp)")
 endif()
 if(DAHENG_INCLUDE_DIR AND DAHENG_LIBRARY)
  target_compile_definitions(${PROJECT_NAME}_input PRIVATE RADAR27_HAS_DAHENG=1)
  target_include_directories(${PROJECT_NAME}_input PRIVATE ${DAHENG_INCLUDE_DIR})
  target_link_libraries(${PROJECT_NAME}_input PRIVATE ${DAHENG_LIBRARY})
  message(STATUS "CameraSource: Daheng enabled")
 endif()
 if(NOT (HIK_INCLUDE_DIR AND HIK_LIBRARY) AND NOT (DAHENG_INCLUDE_DIR AND DAHENG_LIBRARY))
  message(FATAL_ERROR "BUILD_CAMERA=ON requires MVS or Galaxy SDK headers and libraries")
 endif()
endif()
install(TARGETS ${PROJECT_NAME}_input EXPORT export_${PROJECT_NAME}
 ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)

add_library(detect_node_component SHARED src/detect_node.cpp)
target_include_directories(detect_node_component PRIVATE include)
ament_target_dependencies(detect_node_component rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs)
target_link_libraries(detect_node_component ${PROJECT_NAME}_core ${PROJECT_NAME}_input)
install(TARGETS detect_node_component DESTINATION lib)
add_executable(detect_node src/detect_node.cpp)
target_include_directories(detect_node PRIVATE include)
ament_target_dependencies(detect_node rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs)
target_link_libraries(detect_node ${PROJECT_NAME}_core ${PROJECT_NAME}_input)
install(TARGETS detect_node DESTINATION lib/${PROJECT_NAME})
rclcpp_components_register_nodes(detect_node_component "DetectNode")
target_include_directories(${PROJECT_NAME}_core SYSTEM PUBLIC ${TENSORRT_INCLUDE})
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include")
 install(DIRECTORY include/ DESTINATION include)
 ament_export_include_directories(include)
endif()
ament_export_dependencies(OpenCV Threads yaml-cpp CUDAToolkit rclcpp rclcpp_components radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 enable_testing()
 add_executable(test_video_source tests/test_video_source.cpp)
 target_link_libraries(test_video_source ${PROJECT_NAME}_input)
 add_test(NAME test_video_source COMMAND test_video_source)
 add_executable(test_frame_buffer_recorder tests/test_frame_buffer_recorder.cpp)
 target_link_libraries(test_frame_buffer_recorder ${PROJECT_NAME}_input)
 add_test(NAME test_frame_buffer_recorder COMMAND test_frame_buffer_recorder)
 set_tests_properties(test_frame_buffer_recorder PROPERTIES TIMEOUT 15)

 # Test the production node's input/services using a simulated inference backend.
 # No CUDA runtime or production core is linked into this executable.
 add_executable(test_detect_input tests/test_detect_input.cpp tests/fake_detect_pipeline.cpp src/config.cpp src/draw.cpp)
 target_include_directories(test_detect_input PRIVATE include ${TENSORRT_INCLUDE} ${CUDAToolkit_INCLUDE_DIRS})
 target_compile_definitions(test_detect_input PRIVATE RADAR27_TEST_CONFIG_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../radar27_bringup/config/default")
 ament_target_dependencies(test_detect_input rclcpp radar27_interfaces rm_field sensor_msgs cv_bridge std_msgs std_srvs)
 target_link_libraries(test_detect_input ${PROJECT_NAME}_input yaml-cpp::yaml-cpp)
 add_test(NAME test_detect_input COMMAND test_detect_input)
 set_tests_properties(test_detect_input PROPERTIES TIMEOUT 30)
 # Compile the disabled path independently, even in a camera-enabled build.
 add_executable(test_camera_source_disabled tests/test_camera_source_disabled.cpp src/input/camera_source.cpp)
 target_include_directories(test_camera_source_disabled PRIVATE include ${OpenCV_INCLUDE_DIRS})
 target_link_libraries(test_camera_source_disabled ${OpenCV_LIBS})
 add_test(NAME test_camera_source_disabled COMMAND test_camera_source_disabled)
 if(BUILD_CAMERA AND HIK_INCLUDE_DIR AND HIK_LIBRARY)
  # 真实 SDK 头文件 + 桩实现（extern "C" 假 MV_CC_*），不链接真库、不碰硬件。
  # 被测代码是 rb26SDK 的 hik.cpp 加上 production 的 camera_source.cpp，
  # 因此这里直接编译 hik.cpp 源码，而不是链接 rb26SDK 目标（后者会拉进真库）。
  add_executable(test_camera_source_hik
   tests/test_camera_source_hik.cpp src/input/camera_source.cpp rb26SDK/src/hik/hik.cpp)
  target_compile_definitions(test_camera_source_hik PRIVATE RADAR27_HAS_HIK=1 RB26SDK_HAS_HIK=1)
  target_include_directories(test_camera_source_hik PRIVATE
   include ${OpenCV_INCLUDE_DIRS} rb26SDK/include)
  target_link_libraries(test_camera_source_hik ${OpenCV_LIBS} Threads::Threads)
  add_test(NAME test_camera_source_hik COMMAND test_camera_source_hik)
 endif()
endif()
ament_package()
